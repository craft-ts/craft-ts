import {describe,it,expect} from 'vitest';
import {prototypeRendererSignal} from './interpreter';
import {ɵcraftBatch as craftBatch,ɵcraftWatch as craftWatch,ɵCRAFT_SIGNAL as CRAFT_SIGNAL} from '@craft-ts/core';
describe('private renderer cell contract',()=>{
  it('keeps Object.is semantics, batching and subscription disposal',()=>{
    const value=prototypeRendererSignal<number>(NaN),seen:number[]=[];
    const release=craftWatch(()=>{seen.push(value());});
    value.set(NaN);expect(seen).toHaveLength(1);
    value.set(0);value.set(-0);expect(seen).toHaveLength(3);expect(Object.is(seen[2],-0)).toBe(true);
    craftBatch(()=>{value.set(1);value.update(n=>n+1);});expect(seen).toEqual([NaN,0,-0,2]);
    release.destroy();value.set(3);expect(seen).toHaveLength(4);
    expect(value[CRAFT_SIGNAL]).toBeUndefined();
  });
  it('does not subscribe an effect to the equality read inside set',()=>{
    const written=prototypeRendererSignal(0),trigger=prototypeRendererSignal(0);let runs=0;
    const release=craftWatch(()=>{runs++;written.set(trigger());});
    written.set(42);expect(runs).toBe(1);trigger.set(1);expect(runs).toBe(2);expect(written()).toBe(1);release.destroy();
  });
});
